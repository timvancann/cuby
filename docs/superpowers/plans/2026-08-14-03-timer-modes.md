# Timer Modes Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The Timer tab becomes a real speedcube timer: full-solve mode with WCA random-state scrambles, and a CFOP phase timer with a user-editable phase set — both recording attempts to the shared store.

**Architecture:** The Plan-2 timer core (`createTimer`/`tap`) is reused unchanged; a small mode-agnostic flow reducer drives idle → running → done. Scrambles come from `cubing.js`'s `randomScrambleForEvent("333")`, lazily imported so the case-training path never loads it (§4.3) — cubing runs its search in its own internal worker, which satisfies §4.3's "executed in a web worker" intent without a redundant wrapper worker (nested workers are a known browser pitfall). Attempt persistence is extracted from TrainScreen into `data/attempts.ts` (Plan 2 final-review directive) so both screens share one structured-clone-safe write path.

**Tech Stack:** cubing.js (new runtime dep, lazily imported chunk), everything else as in Plan 2.

**Spec:** `cube-trainer-spec.md` — §3.3 (timer modes), §4.3 (full scrambles), §2 (phase set), §7.3/7.4/7.5. Also closes Plan-2 deferrals: persistence extraction, `pagehide` session close, wake lock on Timer.

## Global Constraints

- All Plan-2 global constraints carry over (TS strict, browser-clean src, core takes timestamps/RNG as params, integer ms).
- Work happens directly on `main` (user preference — no branches). Commit after every green task; never commit red.
- cubing.js must be dynamically imported inside the scramble service only — `npm run build` must show it in a separate chunk, and nothing under `src/core` or the Train path may reference it.
- Attempt rows for timer modes: `mode: 'full' | 'cfop'`, no `caseId`, `scramble` = displayed scramble, splits from the timer core (full: `['solve']`; cfop: the active phase set). Flags: `ok`/`dnf` only in the UI (the `Flag` type stays as is; `misrecognized` is case-training-specific).
- Session `configSnapshot`: `[]` for full, the phase labels for cfop (§2).
- §7.3 applies to Timer wholesale: whole-screen tap zone, small no-confirm abort, 300 ms dead-time after the stopping tap, wake lock while mounted, vibration on state advances only.
- Deviation (approved in plan review): v1 ships ONE editable phase set, not multiple named sets; multiple sets go to the §8 roadmap.

`src/ui/phaseSet.ts`:

```ts
export function parsePhaseLabels(text: string): { labels: string[] } | { error: string } {
  const labels = text.split(/[,→>]/).map(s => s.trim()).filter(Boolean);
  if (labels.length === 0) return { error: 'need at least one phase' };
  if (labels.length > 12) return { error: 'at most 12 phases' };
  if (labels.some(l => l.length > 16)) return { error: 'phase names max 16 characters' };
  return { labels };
}
```

```ts
let prefetched: Promise<string> | null = null;

async function generate(): Promise<string> {
  const { randomScrambleForEvent } = await import('cubing/scramble');
  const alg = await randomScrambleForEvent('333');
  return alg.toString();
}

export function warmUp(): void {
  prefetched ??= generate();
}

export function nextFullScramble(): Promise<string> {
  const current = prefetched ?? generate();
  prefetched = generate();
  return current;
}
```

`src/ui/timer/flow.ts`:

```ts
import { createTimer, tap, type TimerState } from '../../core/timer/attempt';

export interface TimerFlow { stage: 'idle' | 'running' | 'done'; timer: TimerState; doneAt: number }
export const DONE_DEAD_MS = 300;

export function newTimerAttempt(phases: readonly string[]): TimerFlow {
  return { stage: 'idle', timer: createTimer(phases), doneAt: 0 };
}

export function tapTimer(s: TimerFlow, now: number): TimerFlow | 'next' {
  if (s.stage === 'idle') return { ...s, stage: 'running', timer: tap(s.timer, now) };
  if (s.stage === 'running') {
    const timer = tap(s.timer, now);
    return timer.status === 'done' ? { stage: 'done', timer, doneAt: now } : { ...s, timer };
  }
  return now - s.doneAt >= DONE_DEAD_MS ? 'next' : s;
}

export function abortTimer(_s: TimerFlow, phases: readonly string[]): TimerFlow {
  return newTimerAttempt(phases);
}
```

```svelte
<script lang="ts">
  import { phaseIndex, splits, totalMs } from '../../core/timer/attempt';
  import { recordAttempt, setAttemptFlag } from '../../data/attempts';
  import { endActiveSession } from '../../data/sessions';
  import { getPhaseSet, getSetting } from '../../data/settings';
  import type { Flag } from '../../data/db';
  import { acquireWakeLock, releaseWakeLock } from '../wakeLock';
  import { nextFullScramble, warmUp } from '../timer/scrambles';
  import { DONE_DEAD_MS, abortTimer, newTimerAttempt, tapTimer, type TimerFlow } from '../timer/flow';

  let mode = $state<'full' | 'cfop'>('full');
  let cfopPhases = $state<string[]>(['Cross', 'F2L', 'OLL', 'PLL']);
  let scramble = $state<string | null>(null);
  let flow = $state<TimerFlow>(newTimerAttempt(['solve']));
  let attemptId = $state<number | null>(null);
  let flag = $state<Flag>('ok');
  let now = $state(0);
  let vibration = $state(true);
  let sessionCount = $state(0);

  const phases = $derived(mode === 'cfop' ? cfopPhases : ['solve']);

  $effect(() => {
    void acquireWakeLock();
    warmUp();
    getPhaseSet().then(p => { cfopPhases = p; });
    getSetting('vibration', true).then(v => { vibration = v; });
    nextFullScramble().then(s => { scramble = s; });
    return () => {
      void releaseWakeLock();
      void endActiveSession(Date.now());
    };
  });

  $effect(() => {
    if (flow.stage !== 'running') return;
    let raf = 0;
    const tick = () => { now = Date.now(); raf = requestAnimationFrame(tick); };
    tick();
    return () => cancelAnimationFrame(raf);
  });

  function fmt(ms: number): string {
    return (ms / 1000).toFixed(2);
  }

  function setMode(m: 'full' | 'cfop') {
    if (flow.stage === 'running') return;
    mode = m;
    flow = newTimerAttempt(m === 'cfop' ? cfopPhases : ['solve']);
    attemptId = null;
  }

  async function onTap() {
    if (!scramble) return;
    const t = Date.now();
    now = t;
    const next = tapTimer(flow, t);
    if (next === flow) return;
    if (vibration && 'vibrate' in navigator) navigator.vibrate(10);
    if (next === 'next') {
      scramble = null;
      flow = newTimerAttempt(phases);
      attemptId = null;
      nextFullScramble().then(s => { scramble = s; });
      return;
    }
    flow = next;
    if (next.stage === 'done') {
      flag = 'ok';
      attemptId = null;
      const id = await recordAttempt({
        mode, config: mode === 'cfop' ? cfopPhases : [], now: t,
        timer: next.timer, scramble,
      });
      attemptId = id;
      sessionCount += 1;
      if (flag !== 'ok') await setAttemptFlag(id, flag);
    }
  }

  function abort() {
    if (flow.stage !== 'running') return;
    flow = abortTimer(flow, phases);
  }

  async function chooseFlag(f: Flag) {
    flag = f;
    if (attemptId !== null) await setAttemptFlag(attemptId, f);
  }
</script>

      <button class:on={mode === 'cfop'} disabled={flow.stage === 'running'} onclick={() => setMode('cfop')}>CFOP</button>
    </div>
```
